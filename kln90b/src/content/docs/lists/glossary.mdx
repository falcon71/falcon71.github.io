import { glossary, glossaryAnchor } from '../../../data/glossary';

The explanations that appear when you point at an underlined term all come from this list.

<dl class="kln-glossary">
	{Object.entries(glossary)
		.sort(([a], [b]) => a.localeCompare(b))
		.map(([term, entry]) => (
			<div id={glossaryAnchor(term)}>
				<dt>
					{term}
					{entry.expansion && <span class="kln-expansion">{entry.expansion}</span>}
				</dt>
				<dd>{entry.text}</dd>
			</div>
		))}
</dl>

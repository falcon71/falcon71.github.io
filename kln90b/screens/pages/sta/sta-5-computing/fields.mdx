<Field id="title" title="Page title">
	`RAIM STATUS`: the page predicts <Term t="RAIM" /> for a destination at a time.
</Field>

<Field id="dest" title="Destination">
	`DEST:` and the identifier of the waypoint that the prediction is for.
</Field>

<Field id="eta" title="Estimated time of arrival">
	`ETA:` and the time of day that the prediction is for, in the time zone below it.
</Field>

<Field id="zone" title="Time zone">
	The system time zone, in which the ETA shows.
</Field>

<Field id="result" title="Calculation in progress">
	`COMPUTING` shows in the line of the bar graph while the unit calculates the prediction. The bars follow when it is done.
</Field>

<Field id="scale" title="Scale of the bar graph">
	The scale under the bars: `-15` under the first bar, `0` under the fourth and `15` under the last. The numbers are minutes before and after the ETA.
</Field>

<Field id="page" title="Page identification">
	The page identification on the left side.
</Field>
